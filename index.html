async function createVideo() { const promptText = document.getElementById("prompt").value; const resultElement = document.getElementById("result"); resultElement.innerText = "Video tayyorlanmoqda..."; const response = await fetch("/create-video/", { method: "POST", headers: {"Content-Type": "application/json"}, body: JSON.stringify({prompt: promptText}) }); const text = await response.text(); // JSON emas, avval matn olamiz try { const data = JSON.parse(text); if (data.video_url) { resultElement.innerText = "Natija: " + data.video_url; } else { resultElement.innerText = "Xato: " + data.error; } } catch (e) { resultElement.innerText = "Server xatosi: " + text; // Server nima yuborsa shuni chiqaramiz } }
